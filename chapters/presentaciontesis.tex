\begin{frame}{Marco Teórico: Problema de Localización}
\phantomsection\label{marco-teuxf3rico-problema-de-localizaciuxf3n}
\justifying

\textbf{Conjuntos:}

\begin{itemize}
\tightlist
\item
  \(\mathcal{I}\): Almacenes candidatos
\item
  \(\mathcal{J}\): Localidades demandantes
\item
  \(\mathcal{P}\): Productos humanitarios
\end{itemize}

\textbf{Variables:}

\begin{itemize}
\tightlist
\item
  \(x_i \in \{0,1\}\): ¿Abrir almacén en \(i\)?
\item
  \(y_{ij} \geq 0\): Flujo \(i \to j\)
\end{itemize}

\textbf{Parámetros clave:}

\begin{itemize}
\tightlist
\item
  \(w_i\): Peso posicional (aptitud logística)
\item
  \(c_{ij}\): Costo unitario de transporte
\item
  \(f_i\): Costo fijo de apertura
\item
  \(d_j\): Demanda esperada en \(j\)
\end{itemize}
\end{frame}

\begin{frame}{Restricciones del Modelo}
\phantomsection\label{restricciones-del-modelo}
\justifying

\textbf{1. Balance de Inventario:}
\[\sum_{j} x_{ij} \leq I_i \quad \forall i\]

\textbf{2. Cobertura de Demanda:}
\[\sum_{i} x_{ij} + z_j = s_j \quad \forall j\]

\textbf{3. Activación Condicional (Big-M):}
\[x_{ij} \leq M \cdot y_i \quad \forall i, j\]

\textbf{4. Capacidad Máxima:} \[I_i \leq C_i \cdot y_i \quad \forall i\]

\vspace{0.3cm}

\alert{Nota:} \(M\) es una constante suficientemente grande.
\end{frame}

\begin{frame}{Componentes del Costo de Inventario}
\phantomsection\label{componentes-del-costo-de-inventario}
\[
TC_{ip} =
{\color{blue}\frac{C^{op}_i D^p_j}{Q^p_{ij}} +
C^{sp}_i \left(\frac{Q^p_{ij}}{2} + SS\right) +
C^p_i D^p_j} {\color{red}+
\frac{D^p_j}{Q^p_{ij}} (C^f_i + \sigma_d E[Z] B)}
\]

\begin{itemize}
\tightlist
\item
  \({\textcolor{blue}{\textbf{Costo de pedido o preparación}}}\): se
  reduce al aumentar la cantidad pedida.\\
\item
  \({\textcolor{blue}{\textbf{Costo del mantenimiento o almacenamiento}}}\):
  considerando el inventario promedio y la seguridad \(SS\).\\
\item
  \({\textcolor{blue}{\textbf{Costo de adquisición}}}\): asociado a la
  demanda esperada \(D^p_j\) del producto \(p\).\\
\item
  \({\textcolor{red}{\textbf{Costo esperado de faltante}}}\): incorpora
  la variabilidad de la demanda \(\sigma_d\) y el costo unitario de
  escasez \(B\).
\end{itemize}
\end{frame}

\begin{frame}{Métricas de Desempeño}
\phantomsection\label{muxe9tricas-de-desempeuxf1o}
\textbf{Fill Rate del Sistema:}
\[FR_{sistema} = \frac{\sum_{i \in I} \sum_{p \in P} D_{ip} \cdot FR_{ip}}{\sum_{i \in I} \sum_{p \in P} D_{ip}}\text{,}\]

\textbf{Costo Total por Persona Atendida:}
\[Costo_{pe} = \frac{Costo_{t}}{Poblacion_{a}}\text{,}\]

\textbf{Tiempo Promedio de Respuesta:}
\[T_{respuesta} = \frac{\sum_{i \in I} \sum_{j \in J} t_{ij} \cdot Y_{ij}}{\sum_{i \in I} \sum_{j \in J} Y_{ij}}\text{,}\]

donde:

\begin{itemize}
\tightlist
\item
  \(t_{ij}\) : tiempo de transporte (horas) desde el almacén \(𝑖\) hasta
  la zona afectada \(𝑗\).
\end{itemize}
\end{frame}

\begin{frame}{¿Por qué L-BFGS-B para este problema?}
\phantomsection\label{por-quuxe9-l-bfgs-b-para-este-problema}
\begin{columns}[T]
\begin{column}{0.5\linewidth}
\textbf{Desafíos del modelo:}

\begin{itemize}
\item Función no convexa y no separable
\item Variables con cotas ($Q \geq 1$, $s_i \geq 0$)
\item Gran escala (miles de variables)
\item Convergencia garantizada
\end{itemize}
\end{column}

\begin{column}{0.5\linewidth}
\textbf{Ventajas de L-BFGS-B:}

\begin{itemize}
\item \textcolor{blue}{Memoria limitada:} $O(mn)$ vs $O(n^2)$
\item \textcolor{green}{Manejo de cotas:} proyección $P_\Omega$
\item \textcolor{red}{Convergencia superlineal}
\item \textcolor{orange}{Eficiencia:} 7 iter. vs 1,183 (Gradiente)
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Método de Solución: Algoritmo L-BFGS-B}
\phantomsection\label{muxe9todo-de-soluciuxf3n-algoritmo-l-bfgs-b}
\begin{columns}[T]
\begin{column}{0.55\linewidth}
\justifying
\setstretch{1.3}

\textbf{¿Por qué este método?}

\begin{itemize}
\item Problema \textbf{MINLP} con variables continuas y cotas ($Q \geq 1$)
\item Función objetivo no convexa globalmente
\item Necesidad de convergencia garantizada
\end{itemize}
\end{column}

\begin{column}{0.45\linewidth}
\justifying
\setstretch{1.3}

\textbf{Fundamento Teórico:}

\begin{itemize}
\item \textbf{Proyección ortogonal:} mantiene factibilidad en cada iteración
\item \textbf{Aproximación del Hessiano:} usa información de curvatura sin calcularla explícitamente
\item \textbf{Condiciones KKT:} garantiza optimalidad bajo restricciones
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Fundamentos Cuasi-Newton}
\phantomsection\label{fundamentos-cuasi-newton}
\justifying

\textbf{Idea central:} Aproximar el Hessiano sin cálculo explícito

\vspace{0.3cm}

\textbf{Ecuación secante:} \[B_{k+1} s_k = y_k\]

donde \(s_k = x_{k+1} - x_k\),
\(y_k = \nabla f(x_{k+1}) - \nabla f(x_k)\)

\vspace{0.3cm}

\textbf{Actualización BFGS:}
\[H_{k+1} = (I - \rho_k s_k y_k^\top) H_k (I - \rho_k y_k s_k^\top) + \rho_k s_k s_k^\top\]
\[\rho_k = (y_k^\top s_k)^{-1}\]

\textbf{Ventajas:}

\begin{itemize}
\item \textbf{Convergencia superlineal:} más rápido que gradiente descendente
\item \textbf{Sin Hessiano explícito:} reduce costo computacional
\item \textbf{Invariancia afín:} independiente del escalado de variables
\end{itemize}

\vspace{0.3cm}

\alert{Condición de curvatura:} \(y_k^\top s_k > 0\) (garantizada por
Wolfe)
\end{frame}

\begin{frame}{Restricciones Tipo Caja y Proyección Ortogonal}
\phantomsection\label{restricciones-tipo-caja-y-proyecciuxf3n-ortogonal}
\begin{columns}[T]
\begin{column}{0.45\linewidth}
\justifying

\textbf{Dominio factible con cotas:}
\[\Omega = \{x \in \mathbb{R}^n \mid \ell_i \leq x_i \leq u_i\}\]

\textbf{Ejemplos en nuestro modelo:}

\begin{itemize}
\item $Q \geq 1$ (al menos una unidad de inventario)
\item $s_i \geq 0$ (inventario no negativo)
\item $y_{ij} \geq 0$ (flujos no negativos)
\end{itemize}

\vspace{0.3cm}

\textbf{Proyección ortogonal:}
\[P_\Omega(z) = \arg\min_{x \in \Omega} \frac{1}{2}\|x - z\|^2\]
\end{column}

\begin{column}{0.45\linewidth}
\justifying

\textbf{Fórmula explícita (componente a componente):}
\[[P_\Omega(z)]_i = \min\{u_i, \max\{\ell_i, z_i\}\}\]

\textbf{Propiedades clave:}

\begin{itemize}
\item \textbf{No expansividad:} $\|P_\Omega(z) - P_\Omega(w)\| \leq \|z-w\|$
\item \textbf{Mantiene factibilidad:} $x_{k+1} \in \Omega$ en cada iteración
\item \textbf{Condiciones KKT:} equivalen a estacionariedad proyectada
\end{itemize}

\vspace{0.3cm}

\alert{Ventaja:} Proyección simple y eficiente (\(O(n)\))
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Condiciones KKT para Problemas con Cotas}
\phantomsection\label{condiciones-kkt-para-problemas-con-cotas}
\justifying

\textbf{Condiciones por componente ($i$):}

\begin{itemize}
\item \textbf{Interior:} $\nabla_i f(x^*) = 0$ si $\ell_i < x^*_i < u_i$
\item \textbf{Cota inferior:} $\nabla_i f(x^*) \geq 0$ si $x^*_i = \ell_i$
\item \textbf{Cota superior:} $\nabla_i f(x^*) \leq 0$ si $x^*_i = u_i$
\end{itemize}

\vspace{0.1cm}

\textbf{Interpretación:} El gradiente debe apuntar
\textbf{hacia adentro} del dominio factible en las fronteras.

\textbf{Estacionariedad Proyectada:}

\[\nabla_\Omega f(x^*) := P_\Omega(x^* - \nabla f(x^*)) - x^* = 0\]

\vspace{0.1cm}

\textbf{Criterio de Convergencia:}

El algoritmo se detiene cuando:
\[\|\nabla_\Omega f(x_k)\| \leq \varepsilon\]
\end{frame}

\begin{frame}{Teorema de Convergencia Global}
\phantomsection\label{teorema-de-convergencia-global}
\justifying

\textbf{Hipótesis (Byrd et al., 1995):}

\begin{itemize}
\item[(H1)] Gradiente Lipschitz-continuo en $\Omega$
\item[(H2)] Conjunto de nivel inicial compacto
\item[ ] Búsqueda lineal satisface condiciones de Wolfe
\end{itemize}

\vspace{0.1cm}

\textbf{Garantía teórica:} El algoritmo no diverge y encuentra puntos
estacionarios factibles.

\textbf{Resultado (Teorema 6.2):}
\[\liminf_{k \to \infty} \|\nabla_\Omega f(x_k)\| = 0\]

\textbf{Interpretación:} Toda subsucesión convergente satisface las
condiciones \textbf{KKT}.

\alert{Importancia:} Valida que la solución numérica obtenida es óptima
bajo restricciones de caja.
\end{frame}
